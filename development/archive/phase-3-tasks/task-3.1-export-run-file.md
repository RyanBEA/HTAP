Upgrades_START

Upgrades_END
```

**Key sections:**
1. **RunParameters**: Configuration settings
2. **RunScope**: Archetypes, locations, rulesets
3. **Upgrades**: Selected options per category

"""
HTAP .run file generator
Creates valid .run files from configuration
"""

from typing import Dict, Set, List, Optional
from datetime import datetime
from pathlib import Path


class RunFileGenerator:
    """
    Generates HTAP .run configuration files
    """

    DEFAULT_PARAMS = {
        'run-mode': 'mesh',
        'archetype-dir': 'C:/HTAP',
        'options-file': 'HTAP-options.json',
        'unit-costs-db': 'HTAPUnitCosts.json',
        'output-folder': './output'
    }

    def __init__(self):
        """Initialize generator with default parameters"""
        self.params = self.DEFAULT_PARAMS.copy()

    def set_parameter(self, key: str, value: str):
        """
        Set a run parameter

        Args:
            key: Parameter name (e.g., 'run-mode')
            value: Parameter value
        """
        self.params[key] = value

    def generate_run_file(
        self,
        run_config: Dict,
        selected_options: Dict[str, Set[str]],
        custom_params: Optional[Dict[str, str]] = None
    ) -> str:
        """
        Generate complete .run file content

        Args:
            run_config: Dict with keys: archetypes, location, ruleset, cost_source
            selected_options: Dict mapping category -> set of choices
            custom_params: Optional custom RunParameters

        Returns:
            String content of .run file
        """
        lines = []

        # Add header comment
        lines.append(f"! HTAP Run Configuration")
        lines.append(f"! Generated by HTAP Configuration Editor")
        lines.append(f"! Date: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        lines.append("")

        # Merge custom parameters
        if custom_params:
            params = {**self.params, **custom_params}
        else:
            params = self.params

        # RunParameters section
        lines.extend(self._generate_run_parameters(params))
        lines.append("")

        # RunScope section
        lines.extend(self._generate_run_scope(run_config))
        lines.append("")

        # Upgrades section
        lines.extend(self._generate_upgrades(selected_options))

        return "\n".join(lines)

    def _generate_run_parameters(self, params: Dict[str, str]) -> List[str]:
        """Generate RunParameters section"""
        lines = ["RunParameters_START"]

        for key, value in params.items():
            lines.append(f"  {key} = {value}")

        lines.append("RunParameters_END")
        return lines

    def _generate_run_scope(self, run_config: Dict) -> List[str]:
        """Generate RunScope section"""
        lines = ["RunScope_START"]

        # Archetypes
        archetypes = run_config.get('archetypes', [])
        if archetypes:
            archetypes_str = ", ".join(archetypes)
            lines.append(f"  archetypes = {archetypes_str}")

        # Location
        location = run_config.get('location')
        if location:
            lines.append(f"  locations = {location}")

        # Ruleset
        ruleset = run_config.get('ruleset', 'as-found')
        lines.append(f"  rulesets = {ruleset}")

        lines.append("RunScope_END")
        return lines

    def _generate_upgrades(self, selected_options: Dict[str, Set[str]]) -> List[str]:
        """Generate Upgrades section"""
        lines = ["Upgrades_START"]

        # Sort categories for consistent output
        for category in sorted(selected_options.keys()):
            choices = selected_options[category]
            if choices:
                # Convert set to sorted list, then join
                choices_str = ", ".join(sorted(choices))
                lines.append(f"  {category} = {choices_str}")

        lines.append("Upgrades_END")
        return lines

    def save_to_file(
        self,
        content: str,
        filepath: str
    ):
        """
        Save .run file to disk

        Args:
            content: Generated .run file content
            filepath: Path to save file
        """
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)

        with open(path, 'w', encoding='utf-8') as f:
            f.write(content)

    def validate_format(self, content: str) -> tuple[bool, List[str]]:
        """
        Validate .run file format

        Args:
            content: .run file content to validate

        Returns:
            Tuple of (is_valid, list_of_errors)
        """
        errors = []
        lines = content.split('\n')

        # Check for required sections
        has_run_params = False
        has_run_scope = False
        has_upgrades = False

        for line in lines:
            if 'RunParameters_START' in line:
                has_run_params = True
            if 'RunScope_START' in line:
                has_run_scope = True
            if 'Upgrades_START' in line:
                has_upgrades = True

        if not has_run_params:
            errors.append("Missing RunParameters section")
        if not has_run_scope:
            errors.append("Missing RunScope section")
        if not has_upgrades:
            errors.append("Missing Upgrades section")

        # Check for matching START/END tags
        if has_run_params:
            if 'RunParameters_END' not in content:
                errors.append("RunParameters_START without RunParameters_END")

        if has_run_scope:
            if 'RunScope_END' not in content:
                errors.append("RunScope_START without RunScope_END")

        if has_upgrades:
            if 'Upgrades_END' not in content:
                errors.append("Upgrades_START without Upgrades_END")

        return len(errors) == 0, errors

"""
Export configuration UI widget
"""

import streamlit as st
from pathlib import Path
from datetime import datetime

from src.utils.run_file_generator import RunFileGenerator, RunFileTemplate
from src.ui.state_manager import get_total_combinations


def render_export_widget():
    """
    Render export configuration widget in RIGHT panel
    """
    st.subheader("💾 Export Configuration")

    # Check if configuration is ready
    run_config = st.session_state.get('run_config', {})
    selected_options = st.session_state.get('selected_options', {})

    if not run_config.get('archetypes'):
        st.warning("⚠️ No archetypes selected")
        return

    if not selected_options:
        st.info("💡 Select some options before exporting")

    # Show summary
    total_combos = get_total_combinations()

    col1, col2 = st.columns(2)
    with col1:
        st.metric("Archetypes", len(run_config.get('archetypes', [])))
        st.metric("Categories", len(selected_options))
    with col2:
        st.metric("Total Runs", total_combos)

        if total_combos > 100:
            st.caption("⚠️ Large run")
        elif total_combos > 0:
            st.caption("✅ Ready")

    st.divider()

    # Template selector
    st.subheader("📋 Template")

    template_names = RunFileTemplate.get_template_names()
    selected_template = st.selectbox(
        "Select template",
        options=["Custom"] + template_names,
        help="Choose a predefined template or customize parameters"
    )

    # Get template params or use defaults
    if selected_template != "Custom":
        template = RunFileTemplate.get_template(selected_template)
        if template:
            st.caption(f"ℹ️ {template['description']}")
            custom_params = template['params']
    else:
        custom_params = None

    # Advanced parameters (expandable)
    with st.expander("⚙️ Advanced Parameters", expanded=False):
        run_mode = st.selectbox(
            "Run mode",
            options=["mesh", "parametric", "sample"],
            index=0,
            help="Mesh = all combinations, Parametric = sequential, Sample = random subset"
        )

        archetype_dir = st.text_input(
            "Archetype directory",
            value="C:/HTAP",
            help="Path to directory containing .h2k archetype files"
        )

        output_folder = st.text_input(
            "Output folder",
            value="./output",
            help="Where to save simulation results"
        )

        # Override template if custom
        if selected_template == "Custom":
            custom_params = {
                "run-mode": run_mode,
                "archetype-dir": archetype_dir,
                "output-folder": output_folder,
                "options-file": "HTAP-options.json",
                "unit-costs-db": "HTAPUnitCosts.json"
            }

    st.divider()

    # Generate preview
    st.subheader("👁️ Preview")

    generator = RunFileGenerator()
    run_content = generator.generate_run_file(
        run_config,
        selected_options,
        custom_params
    )

    # Validate
    is_valid, errors = generator.validate_format(run_content)

    if not is_valid:
        st.error("❌ Generated file has errors:")
        for error in errors:
            st.error(f"  • {error}")

    # Show preview
    with st.expander("📄 Preview .run file", expanded=False):
        st.code(run_content, language="text")

    st.divider()

    # Export buttons
    st.subheader("💾 Export")

    col1, col2 = st.columns(2)

    with col1:
        # Generate filename
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        default_filename = f"htap_config_{timestamp}.run"

        filename = st.text_input(
            "Filename",
            value=default_filename,
            help="Name for the .run file"
        )

    with col2:
        # Download button
        st.download_button(
            label="⬇️ Download .run file",
            data=run_content,
            file_name=filename,
            mime="text/plain",
            use_container_width=True,
            disabled=not is_valid
        )

    # Save to disk button
    if st.button("💾 Save to C:/HTAP", use_container_width=True, disabled=not is_valid):
        try:
            save_path = Path("C:/HTAP") / filename
            generator.save_to_file(run_content, str(save_path))
            st.success(f"✅ Saved to {save_path}")
        except Exception as e:
            st.error(f"❌ Error saving file: {e}")
```
